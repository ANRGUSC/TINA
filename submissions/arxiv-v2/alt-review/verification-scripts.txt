"""Independent recomputation (numpy only; no repo imports, no networkx)."""
import numpy as np
from collections import deque

def bfs_dist(adj):
    n = len(adj); D = np.full((n, n), np.inf)
    for s in range(n):
        D[s, s] = 0; dq = deque([s])
        while dq:
            u = dq.popleft()
            for v in np.flatnonzero(adj[u]):
                if D[s, v] == np.inf:
                    D[s, v] = D[s, u] + 1; dq.append(v)
    return D

def path(n):
    A = np.zeros((n, n), int)
    for i in range(n - 1): A[i, i+1] = A[i+1, i] = 1
    return A
def ring(n):
    A = path(n); A[0, n-1] = A[n-1, 0] = 1; return A
def grid(s):
    A = np.zeros((s*s, s*s), int)
    for i in range(s):
        for j in range(s):
            u = i*s + j
            if i+1 < s: A[u, u+s] = A[u+s, u] = 1
            if j+1 < s: A[u, u+1] = A[u+1, u] = 1
    return A
def geometric(n, seed, radius=0.22):
    rng = np.random.default_rng(seed)
    pos = np.array([rng.random(2) for _ in range(n)])
    E = np.linalg.norm(pos[:, None] - pos[None], axis=2)
    while True:
        A = ((E <= radius) & (E > 0)).astype(int)
        if np.all(np.isfinite(bfs_dist(A))): return A, radius
        radius += 0.02

def cov(A, kappa, nu):
    L = np.diag(A.sum(1)) - A
    w, V = np.linalg.eigh(kappa**2 * np.eye(len(A)) + L)
    S = (V * w**(-nu)) @ V.T
    S /= np.mean(np.diag(S))
    return (S + S.T) / 2

def Kop(D, lc):
    K = np.exp(-D / lc)
    return K / np.linalg.norm(K, axis=1, keepdims=True)

def eta_curve(S, K, D):
    """Residual-operator form: E|| K theta - P theta ||^2 / tr(K S K^T)."""
    tot = np.trace(K @ S @ K.T); out = []
    for r in range(int(D.max()) + 1):
        Rm = K.copy()
        for i in range(len(S)):
            o = np.flatnonzero(D[i] <= r)
            # least-squares weights via Cholesky of observed block
            c = S[o] @ K[i]
            Lc = np.linalg.cholesky(S[np.ix_(o, o)])
            w = np.linalg.solve(Lc.T, np.linalg.solve(Lc, c))
            Rm[i, o] -= w
        out.append(np.trace(Rm @ S @ Rm.T) / tot)
    return np.array(out)

def regret(eta, lat, T):
    r = np.arange(len(eta))
    return 1 - np.exp(-2 * lat * r / T) * (1 - eta)

def direct_delayed_regret(S, K, D, r, lat, T):
    """Brute force: optimal linear rule on the joint (past,current) Gaussian, no composition law."""
    n = len(S); a = np.exp(-lat * r / T)
    J = np.block([[S, a * S], [a * S, S]])   # (past, current)
    tot = np.trace(K @ S @ K.T); err = 0.0
    for i in range(n):
        o = np.flatnonzero(D[i] <= r)           # indices in past block
        tgt = np.concatenate([np.zeros(n), K[i]])  # x*_i = K_i theta_current
        c = J[o] @ tgt
        w = np.linalg.solve(J[np.ix_(o, o)], c)
        e = tgt.copy(); e[o] -= w
        err += e @ J @ e
    return err / tot

def all_policy_optimum(states, pi, q, k, observations, transition):
    """Complete indicator basis, one independent table per agent."""
    columns = []
    for agent, observed in enumerate(observations):
        values = np.unique(states[:, observed], axis=0) if observed else np.zeros((1, 0))
        for value in values:
            col = np.zeros((len(states), len(observations)))
            col[:, agent] = np.all(states[:, observed] == value, axis=1)
            columns.append(col)
    basis = np.stack(columns, axis=2)
    targets = states @ k.T
    joint = pi[:, None] * transition
    hessian = np.einsum('s,sip,ij,sjq->pq', pi, basis, q, basis)
    rhs = np.einsum('st,sip,ij,tj->p', joint, basis, q, targets)
    coefficients = np.linalg.solve(hessian, rhs)
    action = np.einsum('sip,p->si', basis, coefficients)
    error = targets[None, :, :] - action[:, None, :]
    regret = .5 * np.einsum('st,sti,ij,stj->', joint, error, q, error)
    return float(regret), action

"""Brute-force check of the composition law on finite-state, non-Gaussian chains.

State theta in {-1,0,1}^n (n agents on a path graph, scalar local states), plus
optional vector actions. Transition kernel P engineered so that
E[theta' | theta] = a * theta exactly, with innovation law depending on theta.
Team optimum over ALL (nonlinear) rules = weighted least squares over indicator
bases of each agent's observed pattern.
"""
import itertools
import numpy as np
from scipy.optimize import linprog

rng = np.random.default_rng(20261112)


def make_states(n, vals=(-1.0, 0.0, 1.0)):
    return np.array(list(itertools.product(vals, repeat=n)))


def cond_mean_kernel(S, a, n_mix=4, floor=0.0):
    """Row-stochastic P with sum_j P_ij S_j = a S_i; rows are mixtures of LP
    vertices under random costs (state dependent innovations)."""
    m, n = S.shape
    P = np.zeros((m, m))
    A_eq = np.vstack([S.T, np.ones(m)])
    for i in range(m):
        b_eq = np.concatenate([a * S[i], [1.0]])
        row = np.zeros(m)
        wts = rng.dirichlet(np.ones(n_mix))
        for k in range(n_mix):
            c = rng.normal(size=m) + 0.5 * rng.normal() * (S @ S[i])  # state dependent
            res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=[(floor, 1)] * m, method="highs")
            assert res.status == 0, res.message
            row += wts[k] * res.x
        P[i] = np.clip(row, 0, None)
        P[i] /= P[i].sum()
    return P


def stationary(P):
    w, V = np.linalg.eig(P.T)
    k = np.argmin(abs(w - 1))
    pi = np.real(V[:, k])
    pi = pi / pi.sum()
    assert np.all(pi > -1e-12)
    pi = np.clip(pi, 0, None)
    return pi / pi.sum()


def neighborhoods(n, r):
    return [[j for j in range(n) if abs(i - j) <= r] for i in range(n)]


def team_opt(obs_states, target, weights, nbrs, Q, dims):
    """min over u_i = g_i(obs_states[:, nbrs[i]]) of sum_w (u - target)^T Q (u - target).
    obs_states: (N, n) observed state per sample; target: (N, d); weights: (N,).
    dims: action dimension per agent. Returns the minimum value."""
    N = obs_states.shape[0]
    d = sum(dims)
    keep = weights > 0
    obs_states, target, weights = obs_states[keep], target[keep], weights[keep]
    N = obs_states.shape[0]
    L = np.linalg.cholesky(Q).T  # Q = L^T L
    cols = []
    off = 0
    blocks = []
    for i, nb in enumerate(nbrs):
        pats, inv = np.unique(obs_states[:, nb], axis=0, return_inverse=True)
        inv = inv.ravel()
        blocks.append((off, dims[i], inv, len(pats)))
        off += dims[i]
    ncol = sum(b[1] * b[3] for b in blocks)
    # design: u (N x d) = sum over columns; build dense (N*d) x ncol in L-coordinates
    X = np.zeros((N, d, ncol))
    c = 0
    for (o, di, inv, npat) in blocks:
        for comp in range(di):
            for p in range(npat):
                X[inv == p, o + comp, c] = 1.0
                c += 1
    sw = np.sqrt(weights)[:, None, None]
    Xl = np.einsum("ab,nbc->nac", L, X) * sw
    yl = (target @ L.T) * np.sqrt(weights)[:, None]
    Xl = Xl.reshape(N * d, ncol)
    yl = yl.reshape(N * d)
    beta, *_ = np.linalg.lstsq(Xl, yl, rcond=None)
    resid = yl - Xl @ beta
    return float(resid @ resid)


def random_Q(d):
    M = rng.normal(size=(d, d))
    return M @ M.T + 0.3 * np.eye(d)


def run_case(n, a, dims, label):
    S = make_states(n)
    m = len(S)
    P = cond_mean_kernel(S, a)
    assert np.allclose(P @ S, a * S, atol=1e-9)
    pi = stationary(P)
    assert np.allclose(pi @ S, 0, atol=1e-9)
    # innovation depends on old state? check conditional variance spread
    cv = (P @ (S ** 2)) - (P @ S) ** 2
    d = sum(dims)
    K = rng.normal(size=(d, n))
    Q = random_Q(d)
    Sigma = (S * pi[:, None]).T @ S
    Rinf = 0.5 * np.trace(K.T @ Q @ K @ Sigma)
    # joint (old,new)
    I, J = np.meshgrid(np.arange(m), np.arange(m), indexing="ij")
    I, J = I.ravel(), J.ravel()
    wj = (pi[:, None] * P).ravel()
    print(f"--- {label}: n={n}, a={a}, dims={dims}, support of pi={np.sum(pi>1e-12)}/{m}, "
          f"cond-var range per coord: [{cv.min():.3f},{cv.max():.3f}], offdiag Q max={np.abs(Q-np.diag(np.diag(Q))).max():.2f}")
    worst = 0
    for r in range(n):
        nb = neighborhoods(n, r)
        Rfresh = 0.5 * team_opt(S, S @ K.T, pi, nb, Q, dims)
        Rdel = 0.5 * team_opt(S[I], S[J] @ K.T, wj, nb, Q, dims)
        pred = (1 - a * a) * Rinf + a * a * Rfresh
        worst = max(worst, abs(Rdel - pred))
        print(f"   r={r}: R_fresh={Rfresh:.8f} eta={Rfresh/Rinf:.5f}  R_delayed(brute)={Rdel:.8f}  "
              f"law={pred:.8f}  diff={Rdel-pred:+.2e}")
    return worst, (S, P, pi, K, Q, Rinf)


if __name__ == "__main__":
    w = []
    w.append(run_case(3, 0.6, [1, 1, 1], "scalar actions")[0])
    w.append(run_case(3, 0.85, [2, 1, 2], "vector actions")[0])
    w.append(run_case(4, 0.4, [1, 1, 1, 1], "four agents")[0])
    w.append(run_case(4, 0.95, [1, 2, 1, 1], "four agents, slow decay")[0])
    print("max abs discrepancy across cases:", max(w))

def team_opt(joint_vals, joint_p, Q, n):
    """Exact optimum over ALL local rules g_i(theta_i) for finite-support theta.
    joint_vals: (m,n) atoms; joint_p: (m,) probs. Unknowns: g_i(v) for each distinct value v of theta_i."""
    idx=[]; off=0; maps=[]
    for i in range(n):
        vals=np.unique(joint_vals[:,i]); maps.append({v:off+k for k,v in enumerate(vals)}); off+=len(vals)
    N=off
    H=np.zeros((N,N)); c=np.zeros(N)
    for a,p in zip(joint_vals,joint_p):
        S=np.zeros((n,N))
        for i in range(n): S[i,maps[i][a[i]]]=1
        H+=p*S.T@Q@S; c+=p*S.T@a      # E[1/2 u'Qu - theta'u]
    g=np.linalg.solve(H,c)
    cost=0.5*g@H@g-c@g
    return g,maps,cost

def eta_numeric(lc, ls, r, h=0.02, Lmult=14.0, sigma2=1.7):
    L = r + Lmult*lc

    S = sigma2*np.exp(-np.abs(u[:,None]-u[None,:])/ls)
    obs = np.abs(u) <= r + 1e-12
    un = ~obs
    var = w@S@w
    Soo = S[np.ix_(obs,obs)]; Suo = S[np.ix_(un,obs)]; Suu = S[np.ix_(un,un)]
    # x* = w_o.theta_o + w_u.theta_u ; cond var = w_u (Suu - Suo Soo^-1 Sou) w_u
    b = Suo.T@w[un]

        red = b@b/Soo[0,0]
    else:
        red = b@np.linalg.solve(Soo + 1e-12*np.eye(obs.sum()), b)
    cv = w[un]@Suu@w[un] - red
    return cv/var, var

def hopdist(A):
    n=len(A); D=np.full((n,n),np.inf)
    for s in range(n):
        D[s,s]=0; q=deque([s])
        while q:
            u=q.popleft()
            for v in np.nonzero(A[u])[0]:
                if D[s,v]==np.inf: D[s,v]=D[s,u]+1; q.append(v)
    return D

def cov(A,kappa,nu):
    L=np.diag(A.sum(1))-A
    w,V=np.linalg.eigh(kappa**2*np.eye(len(A))+L)
    S=(V*w**(-nu))@V.T
    return S/np.mean(np.diag(S))

def Kop(D,lc):
    K=np.exp(-D/lc); return K/np.linalg.norm(K,axis=1,keepdims=True)

def eta_curve(S,K,D):
    n=len(S); diam=int(D.max()); tot=np.trace(K@S@K.T); out=[]
    for r in range(diam+1):

            res+=K[i]@S@K[i]-c@np.linalg.solve(S[np.ix_(O,O)],c)
        out.append(res/tot)
    return np.array(out)

    n=len(Q); cols=[solve(Q,[F(int(i==j)) for i in range(n)]) for j in range(n)]

        out.append((var-c@np.linalg.solve(Soo,c))/var)
    return np.array(out)
ls=1.0; lc=1.5
rs=np.array([0,0.5,1.0,1.5,2.0,3.0])
expcov=lambda d: np.exp(-d/ls)
mat32=lambda d: (1+np.sqrt(3)*d/ls)*np.exp(-np.sqrt(3)*d/ls)
sqexp=lambda d: np.exp(-d**2/(2*ls**2))
expk=lambda u: np.exp(-np.abs(u)/lc)/(2*lc)
gauk=lambda u: np.exp(-u**2/(2*lc**2))/np.sqrt(2*np.pi*lc**2)
boxk=lambda u: (np.abs(u)<=2*lc)/(4*lc)
def report(name,cov,w):
    e=eta(cov,w,rs)
    print(name)
    print("  r        ",rs)
    print("  eta_S(r) ",np.round(e,5))
    print("  eta_S(r)/eta_S(0) ",np.round(e/e[0],5))
    print("  exp(-2r/lc)       ",np.round(np.exp(-2*rs/lc),5))
    with np.errstate(divide='ignore',invalid='ignore'):
        print("  implied decay length -2r/log(ratio):",np.round(-2*rs[1:]/np.log(e[1:]/e[0]),4), " (paper: lc=%.2f)"%lc)
print("closed form eta_0 = lc/(lc+2ls) =",lc/(lc+2*ls))
report("A. exponential cov + exponential kernel (paper's setting)",expcov,expk)
report("B. Matern-3/2 cov + exponential kernel",mat32,expk)

import numpy as np
# 64-node ring of the paper (Fig. decision relevance): Q=I vs off-diagonal Q, team-optimal linear rules.
n=64; T=6.0; lat=0.12; kap=0.75; nu=1.5
idx=np.arange(n)
d=np.abs(idx[:,None]-idx[None,:]); d=np.minimum(d,n-d)
A=(d==1).astype(float); L=np.diag(A.sum(1))-A
w,V=np.linalg.eigh(kap**2*np.eye(n)+L)
S=(V*w**(-nu))@V.T; S/=np.mean(np.diag(S))
def Kop(lc):
    K=np.exp(-d/lc); return K/np.linalg.norm(K,axis=1,keepdims=True)
def eta_team(K,Q,r):
    # u_i = a_i^T theta_{N_r(i)}; U = sum_i e_i a_i^T P_i ; minimise tr((U-K) S (U-K)^T Q)
    nb=[np.where(d[i]<=r)[0] for i in range(n)]
    off=np.cumsum([0]+[len(b) for b in nb]); m=off[-1]
    H=np.zeros((m,m)); g=np.zeros(m)
    QK=Q@K@S   # (Q K S)
    for i in range(n):
        g[off[i]:off[i+1]]=QK[i,nb[i]]
        for j in range(n):
            if Q[i,j]!=0:
                H[off[i]:off[i+1],off[j]:off[j+1]]=Q[i,j]*S[np.ix_(nb[i],nb[j])]
    a=np.linalg.solve(H,g)
    Rinf=0.5*np.trace(K@S@K.T@Q)
    R=Rinf-0.5*a@g
    # agentwise conditional means evaluated under Q
    U=np.zeros((n,n))
    for i in range(n):
        U[i,nb[i]]=np.linalg.solve(S[np.ix_(nb[i],nb[i])],S[nb[i]]@K[i])
    E=U-K; Ragent=0.5*np.trace(E@S@E.T@Q)
    return R/Rinf, Ragent/Rinf
def run(Q,label,rmax=32):
    print(label)
    for lc in (0.7,2,6):
        K=Kop(lc)
        et=[];ea=[]
        for r in range(0,rmax+1):
            t,a=eta_team(K,Q,r); et.append(t); ea.append(a)
        et=np.array(et);ea=np.array(ea); r=np.arange(rmax+1)
        reg_t=1-np.exp(-2*lat*r/T)*(1-et); reg_a=1-np.exp(-2*lat*r/T)*(1-ea)
        print("  lc=%.1f  team-opt r*=%d  (agentwise-rule r*=%d)  eta_S team[0..5]=%s  agentwise[0..5]=%s  max gap=%.4f"%(lc,reg_t.argmin(),reg_a.argmin(),np.round(et[:6],4),np.round(ea[:6],4),np.max(ea-et)))
run(np.eye(n),"Q = I (paper)")
for gam in (0.5,2.0):
    run(np.eye(n)+gam*np.ones((n,n))/1.0,"Q = I + %.1f * 11^T (aggregate coupling)"%gam)
for c in (0.45,-0.45):
    run(np.eye(n)+c*A,"Q = I + %.2f * Adjacency (nearest-neighbour coupling)"%c)
